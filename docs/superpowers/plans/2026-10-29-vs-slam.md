# VS 新一轮砸入 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** PK 模式打开页面与每次清场时，VS 从屏幕外砸下来，撞击时放出冲击环、整屏抖一下；字效与静止时的排版不变。

**Architecture:** 新模块 `slam.js` 用 Web Animations API 播放三段关键帧（文字、冲击环、整屏），关键帧写成常量，用假元素在 Node 里测。VS 的 DOM 拆成黑底框 `.vs-box`、文字 `.vs-text`、冲击环 `.vs-ring` 三层，动画只作用于文字，黑底不动。`app.js` 在字体加载完、排版就绪之后与清场分支各调用一次，开了「减弱动态效果」就不调用。

**Tech Stack:** 原生 HTML/CSS/JavaScript（ES module）、Web Animations API、Node 内置 `node:test`；集成检查用 headless Chrome + CDP。

**Spec:** `docs/superpowers/specs/2026-10-29-vs-slam-design.md`

```js
/**
 * PK 模式的 VS 砸入：VS 从屏幕外砸下来，撞击时放出一圈冲击环，整屏抖一下。
 * 每开始新的一轮播一次，参数见 docs/superpowers/specs/2026-10-29-vs-slam-design.md 第 3 节。
 * 尺寸都用 vmin，各种屏幕上比例一致。
 */

export const SLAM_MS = 560;

// 撞击时刻在整段动画里的位置
const HIT = 0.42;

// 加速落下，撞击时压扁到 0.8 倍，回弹到 1.12 倍，再落定
const TEXT_FRAMES = [
  { transform: 'scale(3.4) rotate(-12deg)', opacity: 0, easing: 'cubic-bezier(.55, 0, 1, .45)' },
  { transform: 'scale(0.8) rotate(3deg)', opacity: 1, offset: HIT, easing: 'ease-out' },
  { transform: 'scale(1.12) rotate(-1deg)', opacity: 1, offset: 0.66, easing: 'ease-in-out' },
  { transform: 'scale(1) rotate(0deg)', opacity: 1 },
];

// 撞击时从 VS 中心出现，边扩散边变细变淡
const RING_FRAMES = [
  { width: '2.2vmin', height: '2.2vmin', borderWidth: '3.3vmin', opacity: 0 },
  { width: '2.2vmin', height: '2.2vmin', borderWidth: '3.3vmin', opacity: 1, offset: HIT },
  { width: '106vmin', height: '106vmin', borderWidth: '0.3vmin', opacity: 0 },
];

// 撞击之后整屏抖几下，幅度逐渐变小
const SHAKE_FRAMES = [
  { transform: 'translate(0, 0)' },
  { transform: 'translate(0, 0)', offset: HIT },
  { transform: 'translate(-1.9vmin, 1.1vmin)', offset: 0.5 },
  { transform: 'translate(1.7vmin, -1.1vmin)', offset: 0.6 },
  { transform: 'translate(-1.1vmin, 0.6vmin)', offset: 0.72 },
  { transform: 'translate(0.6vmin, -0.3vmin)', offset: 0.85 },
  { transform: 'translate(0, 0)' },
];

/**
 * 从头播放一次砸入。上一次还没播完时再调用，新的盖过旧的。
 * 浏览器不支持 element.animate 时什么都不做，VS 静止显示。
 */
export function slamIn({ text, ring, stage }) {
  if (typeof text?.animate !== 'function') return;
  text.animate(TEXT_FRAMES, { duration: SLAM_MS });
  ring.animate(RING_FRAMES, { duration: SLAM_MS, easing: 'ease-out' });
  stage.animate(SHAKE_FRAMES, { duration: SLAM_MS });
}
```

```js
// VS 砸入的检查。用法：node slam-check.mjs <截图目录>
// 前提：本地服务 http://localhost:8080；headless Chrome 开着 9222 调试端口。
// 在页面脚本之前拦截 Element.prototype.animate，记下每段动画的目标与时长，并留住动画对象；
// 同时记下每次 Enter 的事件时间戳，核对成绩仍按事件时间计算。
import { writeFileSync, mkdirSync } from 'node:fs';
import { join } from 'node:path';

const OUT = process.argv[2];
mkdirSync(OUT, { recursive: true });
const CDP = 'http://127.0.0.1:9222';
const BASE = 'http://localhost:8080';
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
setTimeout(() => { console.error('超时'); process.exit(2); }, 120000).unref();
for (let i = 0; i < 100; i++) {
  try { if ((await fetch(`${CDP}/json/version`)).ok) break; } catch {}
  await sleep(200);
}

const SPY = `(() => {
  window.__anim = []; window.__objs = []; window.__keys = []; window.__errors = [];
  const animate = Element.prototype.animate;
  Element.prototype.animate = function (frames, options) {
    window.__anim.push({ who: this.id || this.getAttribute('class'), ms: options && options.duration });
    const a = animate.apply(this, arguments);
    window.__objs.push(a);
    return a;
  };
  addEventListener('keydown', (e) => { if (e.key === 'Enter' && !e.repeat) window.__keys.push(e.timeStamp); }, true);
  addEventListener('error', (e) => window.__errors.push(String(e.message)));
})();`;

async function open(path, { reduce = false } = {}) {
  const t = await (await fetch(`${CDP}/json/new?about:blank`, { method: 'PUT' })).json();
  const ws = new WebSocket(t.webSocketDebuggerUrl);
  await new Promise((r) => (ws.onopen = r));
  let seq = 0;
  const pending = new Map();
  ws.onmessage = (e) => {
    const m = JSON.parse(e.data);
    if (pending.has(m.id)) { pending.get(m.id)(m); pending.delete(m.id); }
  };
  const send = (method, params = {}) => new Promise((res) => {
    const id = ++seq; pending.set(id, res);
    ws.send(JSON.stringify({ id, method, params }));
  });
  const evaluate = async (expression) => {
    const r = await send('Runtime.evaluate', { expression, awaitPromise: true, returnByValue: true });
    if (r.result.exceptionDetails) throw new Error(r.result.exceptionDetails.exception?.description);
    return r.result.result.value;
  };
  await send('Emulation.setDeviceMetricsOverride', { width: 800, height: 360, deviceScaleFactor: 1, mobile: false });
  if (reduce) {
    await send('Emulation.setEmulatedMedia', { features: [{ name: 'prefers-reduced-motion', value: 'reduce' }] });
  }
  await send('Page.enable');
  // 本机 Chrome 里可能还装着旧版 Service Worker，绕过它和 HTTP 缓存
  await send('Network.enable');
  await send('Network.setBypassServiceWorker', { bypass: true });
  await send('Network.setCacheDisabled', { cacheDisabled: true });
  await send('Page.addScriptToEvaluateOnNewDocument', { source: SPY });
  await send('Page.navigate', { url: BASE + path });
  await sleep(1500);
  const key = (type) => send('Input.dispatchKeyEvent',
    { type, key: 'Enter', code: 'Enter', windowsVirtualKeyCode: 13, nativeVirtualKeyCode: 13 });
  return {
    evaluate,
    close: () => fetch(`${CDP}/json/close/${t.id}`),
    enter: async (wait) => { await key('keyDown'); await key('keyUp'); await sleep(wait); },
    shot: async (file) => {
      const r = await send('Page.captureScreenshot', { format: 'png' });
      writeFileSync(join(OUT, file), Buffer.from(r.result.data, 'base64'));
    },
  };
}

const STATE = `({ anim: window.__anim, keys: window.__keys, errors: window.__errors,
  values: [...document.querySelectorAll('.display')].map((s) => s.dataset.value) })`;
const slams = (m) => m.anim.filter((a) => a.ms === 560);
const who = (m) => slams(m).map((a) => a.who).sort().join(',');
const fmt = (ms) => {
  const c = Math.floor(ms / 10);
  return `${String(Math.floor(c / 100)).padStart(2, '0')}.${String(c % 100).padStart(2, '0')}`;
};
let failed = 0;
const check = (name, pass, detail) => {
  if (!pass) failed++;
  console.log(`${pass ? 'OK  ' : 'FAIL'}  ${name}\n      ${detail}`);
};

// PK：打开砸一次；开始、停表、锁定期不砸；清场再砸一次
{
  const p = await open('/?pk');
  const m0 = await p.evaluate(STATE);
  check('打开页面 · 砸一次（文字、冲击环、整屏各一段 560 ms）', who(m0) === 'stage,vs-ring,vs-text', who(m0) || '没有');
  await p.enter(400); await p.enter(100); await p.enter(700); // 红方开始、停表、锁定期内再拍
  await p.enter(400); await p.enter(700); // 蓝方开始、停表
  const m1 = await p.evaluate(STATE);
  check('开始、停表、锁定期 · 不砸', slams(m1).length === 3, `共 ${slams(m1).length / 3} 次`);
  await p.enter(300); // 清场
  const m2 = await p.evaluate(STATE);
  check('清场 · 再砸一次', slams(m2).length === 6, `共 ${slams(m2).length / 3} 次`);
  check('PK · 没有脚本错误', m2.errors.length === 0, m2.errors.join(' / ') || '无');
  await p.close();
}

// 砸入期间拍下开始：照常计时
{
  const p = await open('/?pk');
  await p.enter(300); await p.enter(700); await p.enter(300); await p.enter(700);
  await p.enter(50); // 清场，砸入开始
  await p.enter(300); // 砸入还没播完就拍下开始
  await p.enter(300); // 停表
  const m = await p.evaluate(STATE);
  await p.close();
  const expect = fmt(m.keys[6] - m.keys[5]);
  check('砸入期间开始计时 · 成绩仍按按键事件的时间戳计算', m.values[0] === expect, `显示 ${m.values[0]}，按事件时间算 ${expect}`);
}

// 减弱动态效果：打开与清场都不砸
{
  const p = await open('/?pk', { reduce: true });
  await p.enter(300); await p.enter(700); await p.enter(300); await p.enter(700); await p.enter(300);
  const m = await p.evaluate(STATE);
  await p.close();
  check('减弱动态效果 · 打开与清场都不砸', slams(m).length === 0, `共 ${slams(m).length / 3} 次`);
}

// 普通模式：没有动画
{
  const p = await open('/');
  await p.enter(300); await p.enter(700); await p.enter(300);
  const m = await p.evaluate(STATE);
  await p.close();
  check('普通模式 · 没有动画', m.anim.length === 0, `${m.anim.length} 段`);
}

// 砸入过程的几帧：触发一次清场后，把三段动画暂停在指定时刻截图
{
  const p = await open('/?pk');
  await p.enter(300); await p.enter(700); await p.enter(300); await p.enter(700);
  await p.enter(800); // 清场，等它播完
  for (const ms of [120, 235, 300, 420]) {
```
