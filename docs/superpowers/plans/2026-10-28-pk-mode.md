# PK 模式 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让秒表能上下分屏留住两名选手的成绩，由 `?pk` 进入。

**Architecture:** 显示逻辑目前硬编码为页面上唯一一组元素，先抽成可实例化的单元 `display.js`，再让页面放两格。PK 模式有独立的七状态机，与普通模式的四状态机并存——两者在「停表后再拍一下」这个转移上语义不同（清零重开 vs 换人/清场），强行合并会把两边都写乱。每格的着色由纯函数从状态推导出 `empty/running/done` 三选一，写进 DOM 由 CSS 取用。

**Tech Stack:** 原生 HTML/CSS/JavaScript（ES module）、Node 内置 `node:test`。

**Spec:** `docs/superpowers/specs/2026-10-27-pk-mode-design.md`

```js
/**
 * PK 模式的状态机。不复用普通模式的四状态机：两者在「停表后再拍一下」
 * 这个转移上语义不同——普通模式是清零重开，PK 模式是换人或清场。
 *
 * WAIT_P1 → RUN_P1 → LOCK_P1 → WAIT_P2 → RUN_P2 → LOCK_P2 → RESULT → WAIT_P1
 *
 * 两个 LOCK 状态各自在 500ms 后由 UNLOCK 事件推进，用途与普通模式一致：
 * 挡掉手掌反弹与多指造成的二次触发。
 */
export const PkState = {
  WAIT_P1: 'WAIT_P1', // 两格皆空，等上格开始
  RUN_P1: 'RUN_P1',   // 上格计时中
  LOCK_P1: 'LOCK_P1', // 上格刚停，吞掉一切输入
  WAIT_P2: 'WAIT_P2', // 上格定格，等下格开始
  RUN_P2: 'RUN_P2',   // 下格计时中
  LOCK_P2: 'LOCK_P2', // 下格刚停，吞掉一切输入
  RESULT: 'RESULT',   // 两格皆定格，等清场
};

export function nextPkState(state, event) {
  if (event === Event.SLAP) {
    switch (state) {
      case PkState.WAIT_P1: return PkState.RUN_P1;
      case PkState.RUN_P1: return PkState.LOCK_P1;
      case PkState.LOCK_P1: return PkState.LOCK_P1;
      case PkState.WAIT_P2: return PkState.RUN_P2;
      case PkState.RUN_P2: return PkState.LOCK_P2;
      case PkState.LOCK_P2: return PkState.LOCK_P2;
      // 只清场，不开跑：现场刚宣布完胜负，一拍就重来会把没记录的成绩抹掉
      case PkState.RESULT: return PkState.WAIT_P1;
      default: return state;
    }
  }
  if (event === Event.UNLOCK) {
    if (state === PkState.LOCK_P1) return PkState.WAIT_P2;
    if (state === PkState.LOCK_P2) return PkState.RESULT;
    return state;
  }
  return state;
}
```

```js
 * 每一格的视觉状态。两种模式都归一到这三个值，CSS 只认它们，
 * 着色规则因此只需写一套。
 */
export const SlotState = {
  EMPTY: 'empty',     // 尚未轮到，只显示暗色幽灵层
  RUNNING: 'running', // 计时中
  DONE: 'done',       // 已定格
};

/** 普通模式：只有一格，锁定与停止在视觉上都是「已定格」。 */
export function slotStateOf(state) {
  if (state === State.RUNNING) return SlotState.RUNNING;
  if (state === State.LOCKED || state === State.STOPPED) return SlotState.DONE;
  return SlotState.EMPTY;
}

/** PK 模式：返回 [上格, 下格]。 */
export function pkSlotStates(state) {
  switch (state) {
    case PkState.RUN_P1:
      return [SlotState.RUNNING, SlotState.EMPTY];
    case PkState.LOCK_P1:
    case PkState.WAIT_P2:
      return [SlotState.DONE, SlotState.EMPTY];
    case PkState.RUN_P2:
      return [SlotState.DONE, SlotState.RUNNING];
    case PkState.LOCK_P2:
    case PkState.RESULT:
      return [SlotState.DONE, SlotState.DONE];
    default:
      return [SlotState.EMPTY, SlotState.EMPTY];
  }
}
```

**Files:**

- [ ] **Step 1: 创建 display.js**

```js
import { formatTime, ghostOf } from './timer-core.js';

/**
 * 把一个 <svg> 包装成独立的显示单元。
 *
 * 两层文字右对齐叠放：幽灵层固定是全 8（七段全亮）显示为暗色，读数层盖在上面。
 * 读数不足两位秒时左边自然露出一个全暗的 8，正是 LED 牌没通电那一位的样子。
 */
export function createDisplay(svg) {
  const ghost = svg.querySelector('.ghost');
  const value = svg.querySelector('.value');

  let lastText = null;
  let lastGhost = null;

  // 字号写死在 CSS 里只是个基准值，真实大小由 viewBox 决定。
  // 这里按幽灵层量出来的实际墨水范围设 viewBox，SVG 自己等比缩放填满容器，
  // 不用手算 vw/vh，各种屏幕比例都能占满。
  function fit() {
    let box;
    try {
      box = ghost.getBBox();
    } catch {
      return; // 元素不可见时部分浏览器会抛错（普通模式下第二格是 display:none）
    }
    if (!box.width) return; // 字体还没加载完，量不出东西
    const pad = box.height * 0.06;
    svg.setAttribute(
      'viewBox',
      `${box.x - pad} ${box.y - pad} ${box.width + pad * 2} ${box.height + pad * 2}`
    );
  }

  function render(text) {
    if (text === lastText) return;
    lastText = text;
    svg.dataset.value = text; // 挂一份明文，方便测试和调试读取
    value.textContent = text;

    const g = ghostOf(text);
    if (g !== lastGhost) { // 只有位数变了才需要重新量
      lastGhost = g;
      ghost.textContent = g;
      fit();
    }
  }

  render(formatTime(0));

  return { render, fit };
}
```

```js
import {
  formatTime, nextState, nextPkState, State, PkState, Event,
  SlotState, pkSlotStates, slotStateOf,
} from './timer-core.js';
import { createDisplay } from './display.js';

const LOCK_MS = 500; // 停表后的封锁时长，防手掌反弹与多指造成的二次触发

const params = new URLSearchParams(location.search);
const isPk = params.has('pk');

// ?green：停表变绿，便于一眼看出表已经停了。PK 模式不吃这个参数。
// 在 iPhone 上用带参数的地址「添加到主屏幕」，那个图标就固定带该设置。
if (params.has('green')) document.documentElement.dataset.greenStop = '';
if (isPk) document.documentElement.dataset.pk = '';

const stage = document.getElementById('stage');
const slots = [...stage.querySelectorAll('.slot')];
const displays = slots.map((slot) => createDisplay(slot.querySelector('.display')));

function paintSlots(states) {
  states.forEach((s, i) => { slots[i].dataset.slotState = s; });
}

// ---- 计时核心 ----
// 只负责时间，显示交给当前绑定的那一格。PK 模式下同时只有一格在跑，
// 已定格的那格不会被碰到。

let startAt = 0;
let rafId = null;
let activeDisplay = null;
let unlockTimer = null;
let activePointerId = null;

function tick() {
  activeDisplay.render(formatTime(performance.now() - startAt));
  rafId = requestAnimationFrame(tick);
}

function startTiming(timeStamp, display) {
  startAt = timeStamp;
  activeDisplay = display;
  acquireWakeLock(); // 拍击是明确的用户手势，此时申请必定成功
  display.render(formatTime(0));
  if (rafId === null) rafId = requestAnimationFrame(tick);
}

function stopTiming(timeStamp) {
  if (rafId !== null) {
    cancelAnimationFrame(rafId);
    rafId = null;
  }
  activeDisplay.render(formatTime(timeStamp - startAt));
}
```

- [ ] **Step 2: 替换 `handleSlap` 函数**

```js
// ---- 模式控制 ----
// 时间戳取自事件本身而非 performance.now()，以排除事件派发和 JS 执行的延迟
// ——两者同为相对 performance.timeOrigin 的高精度时间戳。

let state = State.IDLE;

function handleSlapSingle(timeStamp) {
  const next = nextState(state, Event.SLAP);
  if (next === state) return; // LOCKED 状态下被吞掉
  state = next;

  if (next === State.RUNNING) {
    startTiming(timeStamp, displays[0]);
  } else if (next === State.LOCKED) {
    stopTiming(timeStamp);
    clearTimeout(unlockTimer);
    unlockTimer = setTimeout(() => {
      state = nextState(state, Event.UNLOCK);
      paintSlots([slotStateOf(state), SlotState.EMPTY]);
    }, LOCK_MS);
  }

  paintSlots([slotStateOf(state), SlotState.EMPTY]);
}

let pkState = PkState.WAIT_P1;

function handleSlapPk(timeStamp) {
  const next = nextPkState(pkState, Event.SLAP);
  if (next === pkState) return; // 锁定期内被吞掉
  pkState = next;

  if (next === PkState.RUN_P1) {
    startTiming(timeStamp, displays[0]);
  } else if (next === PkState.RUN_P2) {
    startTiming(timeStamp, displays[1]);
  } else if (next === PkState.LOCK_P1 || next === PkState.LOCK_P2) {
    stopTiming(timeStamp);
    clearTimeout(unlockTimer);
    unlockTimer = setTimeout(() => {
      pkState = nextPkState(pkState, Event.UNLOCK);
      paintSlots(pkSlotStates(pkState));
    }, LOCK_MS);
  } else if (next === PkState.WAIT_P1) {
    displays.forEach((d) => d.render(formatTime(0))); // 清场
  }

  paintSlots(pkSlotStates(pkState));
}

const handleSlap = isPk ? handleSlapPk : handleSlapSingle;
```
